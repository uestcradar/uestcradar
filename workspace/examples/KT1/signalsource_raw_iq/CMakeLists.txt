cmake_minimum_required(VERSION 3.16)
project(signalsource_raw_iq LANGUAGES CXX)
include(CTest)
find_package(cycomm_sdk CONFIG REQUIRED)
add_executable(signalsource-raw-iq src/main.cpp)
target_link_libraries(signalsource-raw-iq PRIVATE cycomm::sdk)
target_compile_features(signalsource-raw-iq PRIVATE cxx_std_20)
target_compile_options(signalsource-raw-iq PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
    add_test(NAME raw-iq-source-help COMMAND signalsource-raw-iq --help)
    add_test(NAME raw-iq-source-reject-negative COMMAND signalsource-raw-iq --frames -1)
    add_test(NAME raw-iq-source-reject-empty COMMAND signalsource-raw-iq --samples 0)
    set_tests_properties(raw-iq-source-reject-negative raw-iq-source-reject-empty PROPERTIES WILL_FAIL TRUE)
endif()
