cmake_minimum_required(VERSION 3.16)
project(uestcradar_pcie_source LANGUAGES C CXX)
include(CTest)
find_package(cycomm_sdk CONFIG REQUIRED)
find_package(Threads REQUIRED)

add_library(pcie_json STATIC pcie/cJSON.c)
target_include_directories(pcie_json PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/pcie")

add_library(pcie_data INTERFACE)
target_include_directories(pcie_data INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(pcie_data INTERFACE cycomm::sdk pcie_json)
target_compile_features(pcie_data INTERFACE cxx_std_20)
target_compile_options(pcie_data INTERFACE -Wall -Wextra -Wpedantic)

add_library(pcie_rx STATIC pcie/xdma_rx.c pcie/devmem_io.c)
target_include_directories(pcie_rx PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/pcie")
target_compile_options(pcie_rx PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pcie_source src/main.cpp src/pcie_receiver.cpp)
target_link_libraries(pcie_source PRIVATE pcie_data pcie_rx Threads::Threads)

if(BUILD_TESTING)
    foreach(check iq_decode cpi_templates cpi_assembler options receiver timestamp_check receiver_control dma_snapshot)
        add_executable(${check}_test tests/${check}_test.cpp)
        target_link_libraries(${check}_test PRIVATE pcie_data pcie_rx Threads::Threads)
        target_compile_options(${check}_test PRIVATE -UNDEBUG)
        add_test(NAME pcie-${check} COMMAND ${check}_test)
    endforeach()
    set_tests_properties(pcie-dma_snapshot pcie-receiver_control PROPERTIES TIMEOUT 15)
    target_sources(receiver_control_test PRIVATE src/pcie_receiver.cpp)
    set(CPI_REFERENCE_DATA_ROOT "" CACHE PATH "Optional signalsource CPI0-CPI9 dataset for differential tests")
    if(CPI_REFERENCE_DATA_ROOT)
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../signalsource/src/cpi_data.hpp")
            message(FATAL_ERROR "Reference check requires sibling signalsource source")
        endif()
        add_executable(template_reference_test tests/template_reference_test.cpp)
        target_link_libraries(template_reference_test PRIVATE pcie_data)
        target_include_directories(template_reference_test PRIVATE ../signalsource/src)
        target_compile_options(template_reference_test PRIVATE -UNDEBUG)
        add_test(NAME pcie-template-reference COMMAND template_reference_test "${CPI_REFERENCE_DATA_ROOT}")
    endif()
    add_test(NAME pcie-help COMMAND pcie_source --help)
    add_test(NAME pcie-reject-channel COMMAND pcie_source --capture-only --channel 8)
    set_tests_properties(pcie-reject-channel PROPERTIES WILL_FAIL TRUE)
    add_test(NAME pcie-reject-unimplemented-output COMMAND pcie_source)
    set_tests_properties(pcie-reject-unimplemented-output PROPERTIES WILL_FAIL TRUE)
endif()
